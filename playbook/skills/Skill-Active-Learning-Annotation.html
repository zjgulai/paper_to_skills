1. 解决的问题

数据团队面临"标注预算有限但模型训练数据质量不足"——主动学习以20%标注量达到全量随机标注的模型性能，年化将标注成本从15万元降至3万元节省12万元

2. 核心算法逻辑

核心思想：随机标注数据效率低下——模型最需要的是它"最不确定"的样本，而非随机样本。主动学习让模型主动选择对自身提升最有价值的样本请求人工标注，用 1020% 的标注量达到全量随机标注的效果。

3. 业务应用场景

场景1：婴儿产品评论情感标注（ABSA 方面级情感） - 业务问题：VOC 分析模型需要 5000 条方面级标注评论，专家标注成本约 15 万元，预算仅 3 万元 - 数据要求：未标注评论池（10 万条）+ 初始标注集（200 条）+ 标注界面 - 预期产出：以 20% 的标注量（1000 条）达到 5000 条随机标注的 F1 性能 - 业务价值：标注成本从 15 万降至 3 万，节省 12 万元，且模型性能相当

**三轨验证**： - 成本：需要构建主动学习 pipeline（约 3 人天开发），后续每轮采样自动化 - 合规：标注数据涉及用户评论，需脱敏 PII 后才能送标注员 - 风险：初始标注集质量差会导致主动学习陷入局部最优，需确保初始数据多样性

import numpy as np
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import f1_score

def uncertainty_sampling(model, unlabeled_features: np.ndarray, k: int) -> np.ndarray:
    """选出预测熵最大的 k 个样本索引"""
    proba = model.predict_proba(unlabeled_features)
    # 计算预测熵
    entropy = -np.sum(proba * np.log(proba + 1e-10), axis=1)
    return np.argsort(entropy)[-k:]  # 返回熵最大的 k 个

def run_active_learning(
    all_features: np.ndarray,
    all_labels: np.ndarray,
    initial_n: int = 50,
    query_k: int = 50,
    n_rounds: int = 5,
) -> list:
    """主动学习迭代，返回每轮 F1 分数"""
    np.random.seed(42)
    n = len(all_features)
    # 初始随机标注
    labeled_idx = set(np.random.choice(n, initial_n, replace=False))
    unlabeled_idx = set(range(n)) - labeled_idx
    scores = []
    for round_i in range(n_rounds):
        L_idx = list(labeled_idx)
        U_idx = list(unlabeled_idx)
        model = LogisticRegression(max_iter=300, random_state=42)
        model.fit(all_features[L_idx], all_labels[L_idx])
        # 评估
        f1 = f1_score(all_labels[U_idx], model.predict(all_features[U_idx]),
                      average="macro")
        scores.append((len(labeled_idx), round(f1, 4)))
        # 主动采样
        query_idx = uncertainty_sampling(model, all_features[U_idx], query_k)
        new_labeled = [U_idx[i] for i in query_idx]
        labeled_idx.update(new_labeled)
        unlabeled_idx -= set(new_labeled)
    return scores

def run_random_baseline(all_features, all_labels, initial_n=50,
                        query_k=50, n_rounds=5):
    """随机采样基准对比"""
    np.random.seed(42)
    n = len(all_features)
    labeled_idx = set(np.random.choice(n, initial_n, replace=False))
    unlabeled_idx = set(range(n)) - labeled_idx
    scores = []
    for _ in range(n_rounds):
        L_idx = list(labeled_idx)
        U_idx = list(unlabeled_idx)
        model = LogisticRegression(max_iter=300, random_state=42)
        model.fit(all_features[L_idx], all_labels[L_idx])
        f1 = f1_score(all_labels[U_idx], model.predict(all_features[U_idx]),
                      average="macro")
        scores.append((len(labeled_idx), round(f1, 4)))
        new = np.random.choice(list(unlabeled_idx), min(query_k, len(unlabeled_idx)),
                               replace=False)
        labeled_idx.update(new)
