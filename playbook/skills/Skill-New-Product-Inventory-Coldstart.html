@dataclass
class AnalogSKU:
    sku_id: str
    monthly_sales: list[float]
    unit_price: float
    category: str
    launch_month: int
    price_tier: str

    @property
    def demand_mean(self) -> float:
        return float(np.mean(self.monthly_sales))

    @property
    def demand_std(self) -> float:
        return float(np.std(self.monthly_sales))

    @property
    def cv(self) -> float:
        return self.demand_std / max(self.demand_mean, 1)
