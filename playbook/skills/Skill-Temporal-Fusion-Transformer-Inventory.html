    # 变量重要性计算（模拟 VSN 门控）
    # 使用相关系数的绝对值作为重要性指标
    var_importance = np.zeros(n_feat)
    for i in range(n_feat):
        corr = np.corrcoef(y_hist, exog[:, i])[0, 1]
        var_importance[i] = np.abs(corr) if not np.isnan(corr) else 0.0
    
    # 归一化重要性权重
    var_importance_sum = var_importance.sum()
